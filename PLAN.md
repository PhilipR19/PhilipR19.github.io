> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a lean, publications-forward research-program website for Philip Ruppert at `https://PhilipR19.github.io`, with publications auto-pulled from ORCID and push-to-deploy via GitHub Actions.

**Architecture:** A Quarto website (plain-text `.qmd` + git). **Tech Stack:** Quarto · Python 3.12 (stdlib `urllib` + `json`, no third-party deps) · GitHub Actions · GitHub Pages.

**Where this runs:** A **new standalone repo `PhilipR19.github.io`** — NOT the ScienceOS vault (per `skills/repo-handoff.md`). This plan is the handoff artifact; execute it in a repo-scoped session after creating the empty repo.

**Decision baked in (override if you disagree):** the ORCID fetcher is **Python**, not R — it needs no toolchain in CI and is self-contained. Spec §4 defaulted to R; this is the deviation.

## Task 2: ORCID → citations build script

**Files:**
- Create: `scripts/fetch_orcid.py`, `tests/test_fetch_orcid.py`

The script: (1) GET `pub.orcid.org/v3.0/<id>/works` for work-group summaries; (2) for each, take the preferred summary's DOI; (3) resolve each DOI to CSL-JSON via `https://doi.org/<DOI>` with `Accept: application/vnd.citationstyles.csl+json`; (4) format `Authors. Title. Journal Year. doi`; (5) group by year descending into `_publications.md`. Works without a DOI fall back to ORCID title+year. Bold "Ruppert" in author lists.

- [ ] **Step 3: Write `scripts/fetch_orcid.py`**

```python
#!/usr/bin/env python3
"""Fetch Philip Ruppert's works from ORCID, resolve DOIs to citations,
write a year-grouped Markdown partial (_publications.md).

Stdlib only. Run: python scripts/fetch_orcid.py
"""
from __future__ import annotations
import json
import sys
import urllib.request
from collections import defaultdict
from pathlib import Path

ORCID = "0000-0002-4028-8200"
WORKS_URL = f"https://pub.orcid.org/v3.0/{ORCID}/works"
OUT = Path(__file__).resolve().parent.parent / "_publications.md"
TIMEOUT = 30


def _get(url: str, accept: str) -> bytes:
    req = urllib.request.Request(url, headers={"Accept": accept, "User-Agent": "pr-website/1.0"})
    with urllib.request.urlopen(req, timeout=TIMEOUT) as r:
        return r.read()


def fetch_dois_and_titles() -> list[dict]:
    """Return [{doi, title, year}] from ORCID work-group summaries."""
    data = json.loads(_get(WORKS_URL, "application/json"))
    items = []
    for group in data.get("group", []):
        summary = group["work-summary"][0]
        title = (((summary.get("title") or {}).get("title")) or {}).get("value", "")
        yd = (summary.get("publication-date") or {}).get("year") or {}
        year = yd.get("value") if yd else None
        doi = None
        for eid in ((summary.get("external-ids") or {}).get("external-id") or []):
            if eid.get("external-id-type", "").lower() == "doi":
                doi = (eid.get("external-id-value") or "").strip()
                break
        items.append({"doi": doi, "title": title, "year": year})
    return items


def bold_author(name: str) -> str:
    return f"**{name}**" if "ruppert" in name.lower() else name


def _author_list(csl: dict) -> str:
    names = []
    for a in csl.get("author", []) or []:
        given, family = a.get("given", ""), a.get("family", "")
        full = (f"{given} {family}").strip() or a.get("literal", "")
        if full:
            names.append(bold_author(full))
    if len(names) > 8:
        names = names[:8] + ["et al."]
    return ", ".join(names)


def format_citation(csl: dict) -> str:
    authors = _author_list(csl)
    title = (csl.get("title") or "").rstrip(".")
    journal = csl.get("container-title", "")
    year = ""
    issued = (csl.get("issued") or {}).get("date-parts") or []
    if issued and issued[0]:
        year = str(issued[0][0])
    doi = csl.get("DOI", "")
    parts = []
    if authors:
        parts.append(authors + ".")
    parts.append(f"*{title}*.")
    if journal:
        parts.append(f"{journal}")
    if year:
        parts.append(f"{year}.")
    line = " ".join(parts)
    if doi:
        line += f" [https://doi.org/{doi}](https://doi.org/{doi})"
    return line


def resolve(doi: str) -> dict | None:
    try:
        raw = _get(f"https://doi.org/{doi}", "application/vnd.citationstyles.csl+json")
        return json.loads(raw)
    except Exception as e:  # noqa: BLE001
        print(f"  ! DOI resolve failed for {doi}: {e}", file=sys.stderr)
        return None


def year_of(csl_or_item) -> int:
    issued = (csl_or_item.get("issued") or {}).get("date-parts") or []
    if issued and issued[0]:
        return int(issued[0][0])
    y = csl_or_item.get("year")
    return int(y) if y else 0


def main() -> int:
    items = fetch_dois_and_titles()
    print(f"ORCID returned {len(items)} work groups", file=sys.stderr)
    by_year: dict[int, list[str]] = defaultdict(list)
    for it in items:
        csl = resolve(it["doi"]) if it["doi"] else None
        if csl:
            by_year[year_of(csl)].append(format_citation(csl))
        else:
            # fallback: title + year only, no citation metadata
            yr = int(it["year"]) if it["year"] else 0
            line = f"*{(it['title'] or 'Untitled').rstrip('.')}*. {it['year'] or ''}".strip()
            by_year[yr].append(line)
    lines = ["<!-- generated by scripts/fetch_orcid.py — do not edit by hand -->", ""]
    for yr in sorted(by_year, reverse=True):
        header = str(yr) if yr else "Other"
        lines.append(f"### [{header}]{{.pub-year}}")
        lines.append("")
        for c in by_year[yr]:
            lines.append(f"- {c}")
        lines.append("")
    OUT.write_text("\n".join(lines))
    print(f"wrote {OUT} ({sum(len(v) for v in by_year.values())} entries)", file=sys.stderr)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

## Task 3: GitHub Actions — render + deploy to Pages

**Files:**
- Create: `.github/workflows/publish.yml`

- [ ] **Step 1: Write `.github/workflows/publish.yml`**

```yaml
name: publish
on:
  push:
    branches: [main]
  workflow_dispatch:
  schedule:
    - cron: "0 6 * * 1"   # weekly rebuild so new ORCID entries appear

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-python@v6
        with:
          python-version: "3.12"
      - name: Refresh publications from ORCID (non-fatal)
        run: python scripts/fetch_orcid.py || echo "ORCID fetch failed; using committed snapshot"
      - uses: quarto-dev/quarto-actions/setup@v2
      - uses: quarto-dev/quarto-actions/render@v2
      - uses: actions/upload-pages-artifact@v3
        with:
          path: _site

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
```
